select c.customer_name,o.order_id,c.customer_id,o.product_name,o.order_date,o.quantity from customers c
inner join orders o 
on c.customer_id=o.customer_id;

select p.product_name,c.category_name from products p 
join categories c 
on p.category_id=c.category_id;

select c.category_name,p.product_name,p.price from products p
 join categories c
 on p.category_id=c.category_id;
